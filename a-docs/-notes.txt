import Cocoa

@main
class AppDelegate: NSObject, NSApplicationDelegate {
    
    func applicationDidFinishLaunching(_ aNotification: Notification) {
        // App setup
    }
    
    // Handle URL when app is already running
    func application(_ application: NSApplication, open urls: [URL]) {
        for url in urls {
            handleCustomURL(url)
        }
    }
    
    private func handleCustomURL(_ url: URL) {
        print("Received URL: \(url)")
        
        guard url.scheme == "yourapp" else {
            print("Invalid scheme: \(url.scheme ?? "none")")
            return
        }
        
        let components = URLComponents(url: url, resolvingAgainstBaseURL: false)
        let host = url.host ?? ""
        let queryItems = components?.queryItems ?? []
        
        // Convert query items to dictionary
        var params: [String: String] = [:]
        for item in queryItems {
            params[item.name] = item.value ?? ""
        }
        
        // Route based on host
        switch host {
        case "document":
            handleDocumentAction(params: params)
        case "settings":
            handleSettingsAction(params: params)
        case "search":
            handleSearchAction(params: params)
        default:
            handleDefaultAction(url: url)
        }
    }
    
    private func handleDocumentAction(params: [String: String]) {
        if let docName = params["name"] {
            // Open specific document
            print("Opening document: \(docName)")
            // Your document opening logic here
        }
    }
    
    private func handleSettingsAction(params: [String: String]) {
        if let tab = params["tab"] {
            // Open settings to specific tab
            print("Opening settings tab: \(tab)")
            // Your settings opening logic here
        }
    }
    
    private func handleSearchAction(params: [String: String]) {
        if let query = params["q"] {
            // Perform search
            print("Searching for: \(query)")
            // Your search logic here
        }
    }
    
    private func handleDefaultAction(url: URL) {
        print("Default action for: \(url)")
        // Bring app to front by default
        NSApp.activate(ignoringOtherApps: true)
    }
}
